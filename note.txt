use std::{
    sync::{mpsc, Arc, Mutex},
    thread,
    time::Duration,
};

fn main() {
    let worker_count = 5;
    let (sender, receiver) = mpsc::channel::<u32>();

    // Receiver باید میان workerها به اشتراک گذاشته شود.
    let receiver = Arc::new(Mutex::new(receiver));

    let mut workers = Vec::with_capacity(worker_count);

    for worker_id in 0..worker_count {
        let receiver = Arc::clone(&receiver);

        let handle = thread::spawn(move || {
            loop {
                // فقط برای دریافت یک job قفل را می‌گیریم؛
                // خودِ کار بیرون از lock اجرا می‌شود.
                let item = {
                    let receiver = receiver.lock()
                        .expect("receiver mutex was poisoned");

                    receiver.recv()
                };

                match item {
                    Ok(item) => {
                        thread::sleep(Duration::from_secs(5));
                        println!("worker {worker_id}: I am number {item}");
                    }

                    // وقتی همهٔ Senderها حذف شوند، کانال بسته می‌شود.
                    Err(_) => break,
                }
            }
        });

        workers.push(handle);
    }

    // ارسال ۱۰۰ job؛ ایجاد ۱۰۰ ترد رخ نمی‌دهد.
    for item in 0..100 {
        sender.send(item).expect("all workers disconnected");
    }

    // با حذف sender کانال بسته می‌شود و workerها پس از اتمام jobها خارج می‌شوند.
    drop(sender);

    for worker in workers {
        worker.join().expect("a worker thread panicked");
    }

    println!("\nThe End ...\n");
}
